async function mutation(
  w: ConvexTenantWorld,
  fn: FunctionReference<"mutation">,
  args: object,
): Promise<any> {
  const r = (await w.t
    .withIdentity(IDENTITY)
    .mutation(fn, args as never)) as TenantFunctionOutcome<any>;
  expect(r.ok, JSON.stringify(r)).toBe(true);
  if (!r.ok) throw new Error("Unexpected authorization denial");
  return r.value;
}

const draft = (w: ConvexTenantWorld) => ({
  warehouseId: w.warehouses.alphaA,
  code: "E001",
  displayName: "Probe Employee",
  employmentStartDate: "2026-01-05",
  status: "ACTIVE",
});
async function hire(
  w: ConvexTenantWorld,
  extra = {},
): Promise<Id<"hrEmployees">> {
  const r = await mutation(w, api.hr.setup.saveEmployee, {
    ...draft(w),
    ...extra,
    requestId: requestId(),
  });
  expect(r.written).toBe(true);
  return r.documentId as Id<"hrEmployees">;
}
async function event(
  w: ConvexTenantWorld,
  id: Id<"hrEmployees">,
  kind: string,
  date: string,
  extra = {},
) {
  const employee = await w.t.run((ctx) => ctx.db.get(id));
  return mutation(w, api.hr.employeeProfile.recordEmploymentEvent, {
    employeeId: id,
    expectedVersion: employee!.version,
    requestId: requestId(),
    kind,
    effectiveDate: date,
    reason: "Hardening scenario",
    ...(kind === "OFFBOARD" ? { separationType: "RESIGNATION" } : {}),
    ...extra,
  });
}
it("independent gap direct detail never invents a missing attendance obligation", async () => {
  const w = await fixture();
  const schedule = { workDays:[1,2,3,4,5,6,7], startTime:"09:00", endTime:"18:00", endsNextDay:false, breakMinutes:60 };
  const selfId = await hire(w,{userId:w.userA,schedule});
  const otherId = await hire(w,{code:"E002",schedule});
  for (const id of [selfId,otherId]) {
    expect((await event(w,id,"OFFBOARD","2026-09-30")).written).toBe(true);
    expect((await event(w,id,"REINSTATE","2026-10-19")).written).toBe(true);
  }
  const self = await query(w,api.hr.self.dayDetail,{businessDate:"2026-10-05"});
  const review = await query(w,api.hr.review.dayDetail,{employeeId:otherId,businessDate:"2026-10-05"});
  const history = await query(w,api.hr.self.history,{from:"2026-10-05",to:"2026-10-05"});
  const rejected = await mutation(w,api.hr.self.submitCorrection,{businessDate:"2026-10-05",start:{minute:540,nextDay:false},end:{minute:1080,nextDay:false},reason:"This gap was not employment",requestId:requestId()});
  const disposition = await mutation(w,api.hr.review.disposeDay,{employeeId:otherId,businessDate:"2026-10-05",expectedRevision:0,disposition:"ABSENT",reason:"This gap was not employment",requestId:requestId()});
  writeFileSync("docs/plans/hr-parity/overnight-evidence/employee-hardening-review-probes/gap-day.json",JSON.stringify({self,review,history,rejected,disposition},null,2)+"\n");
  expect(history.rows).toEqual([]);
  expect(rejected).toMatchObject({written:false,error:{code:"HR_OUTSIDE_EMPLOYMENT"}});
  expect(disposition).toMatchObject({written:false,error:{code:"HR_OUTSIDE_EMPLOYMENT"}});
  expect(self.detail.issue).not.toBe("MISSING_RECORD");
  expect(review.detail.issue).not.toBe("MISSING_RECORD");
});
