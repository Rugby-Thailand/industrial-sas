// Generates the HR feature matrix from the research slice specs plus audited overrides.
// Usage: node /tmp/kiro-hr-513a/matrix.mjs <researchRoot> <outDir>
import { readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const [researchRoot, outDir] = process.argv.slice(2);
const slices = readdirSync(researchRoot).filter((d) => /^0\d-/.test(d)).sort();

const rows = [];
for (const slice of slices) {
  const spec = readFileSync(join(researchRoot, slice, "spec.md"), "utf8");
  for (const line of spec.split("\n")) {
    const m = /^\|\s*(S0\d-\d+)\s*\|/.exec(line);
    if (!m) continue;
    const cells = line.split("|").map((c) => c.trim());
    rows.push({
      id: cells[1],
      task: cells[2],
      label: cells[4],
      haveToday: cells[6],
      slice,
    });
  }
}

const baselineStatus = (row) => {
  const have = row.haveToday.toLowerCase();
  if (row.label === "unknown" && have.startsWith("miss")) return "blocked";
  if (have.startsWith("have")) return "delivered";
  if (have.startsWith("partial")) return "partial";
  return "missing";
};

const out = rows.map((row) => {
  const o = O[row.id];
  const remainingDefault =
    row.slice.startsWith("08") && baselineStatus(row) !== "delivered"
      ? "Payroll calculation engine missing"
      : row.haveToday;
  return {
    feature: `${row.id} ${row.task}`,
    slice: row.slice,
    implementationStatus: o ? o[0] : baselineStatus(row),
    verificationLevel: o ? o[1] : T.baseline,
    evidence: o
      ? `${o[2]} (research label ${row.label}; spec ${row.slice}/spec.md)`
      : `Research 'We have today': ${row.haveToday} (label ${row.label}; spec ${row.slice}/spec.md)`,
    remaining: o ? o[3] : remainingDefault,
  };
});
