import {createRoot} from 'react-dom/client';
import {useState} from 'react';
import {NextIntlClientProvider} from 'next-intl';
import {NavigationGuardProvider,useNavigationDraft} from '@/components/navigation/NavigationGuardProvider';
import {ConfirmDialog} from '@/components/ui/ConfirmDialog';
import {Button} from '@/components/ui/button';
import './style.css';
const params=new URLSearchParams(location.search);
const locale=params.get('locale')==='th'?'th':'en';
const thai=locale==='th';
document.documentElement.lang=locale;
document.documentElement.classList.toggle('dark',params.get('theme')==='dark');
function GuardFixture(){
 const [draft,setDraft]=useState('Draft value');const [continued,setContinued]=useState(0);
 const guard=useNavigationDraft({dirty:!!draft,pending:params.get('pending')==='1',discard:()=>setDraft('')});
 return <><label>Draft<input value={draft} onChange={e=>setDraft(e.target.value)}/></label><Button onClick={()=>guard.requestTransition(()=>setContinued(x=>x+1))}>Open guard</Button><output data-testid="continued">{continued}</output></>;
}
function ConfirmFixture(){
 const [open,setOpen]=useState(false);const [confirmed,setConfirmed]=useState(0);
 return <><Button onClick={()=>setOpen(true)}>Open confirmation</Button><output data-testid="confirmed">{confirmed}</output><ConfirmDialog open={open} onOpenChange={setOpen} title={thai?'ลบใบ Job?':'Delete job ticket?'} description={thai?'ใบ Job สำหรับการทดสอบ':'Synthetic job ticket for component verification'} cancelLabel={thai?'ยกเลิก':'Cancel'} confirmLabel={thai?'ลบ':'Delete'} pendingLabel={thai?'กำลังลบ…':'Deleting…'} tone="destructive" pending={params.get('pending')==='1'} onConfirm={()=>setConfirmed(x=>x+1)}/></>;
}
createRoot(document.getElementById('root')!).render(<NextIntlClientProvider locale={locale} messages={{}} timeZone="Asia/Bangkok"><main className="p-6"><h1>Local confirmation verification</h1>{params.get('kind')==='guard'?<NavigationGuardProvider><GuardFixture/></NavigationGuardProvider>:<ConfirmFixture/>}</main></NextIntlClientProvider>);
